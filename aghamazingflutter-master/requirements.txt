# Flutter Development Environment Requirements for Windows
# This file outlines the tools and versions needed to develop the Flutter app on Windows

# Flutter SDK
# Download from: https://docs.flutter.dev/get-started/install/windows
# Current stable version: 3.16.0 or higher

# Dart SDK
# Included with Flutter SDK installation

# Android Studio / Android SDK
# Minimum version: Android Studio Flamingo | 2022.2.1
# Android SDK Platform 33 (API level 33)
# Android SDK Command-line Tools
# Android SDK Build-Tools 33.0.2

# Java Development Kit
# Minimum version: OpenJDK 17 or Oracle JDK 17
# Ensure JAVA_HOME environment variable is set

# Git for Windows
# Version: 2.40 or higher
# Required for Flutter packages and version control

# Node.js (optional, for Firebase tools)
# Version: 16.0 or higher

# Firebase CLI (optional, for Firebase functions)
# Install via npm: npm install -g firebase-tools

# Windows-specific dependencies:
# Windows PowerShell 5.0 or later
# Windows Command Prompt (cmd.exe) or Windows Terminal
# Visual Studio Build Tools (for native plugins compilation)

# Environment Variables for Windows:
# ANDROID_HOME=C:\Users\[USERNAME]\AppData\Local\Android\Sdk
# JAVA_HOME=C:\Program Files\Java\jdk-17
# FLUTTER_ROOT=C:\flutter
# PATH=%PATH%;%FLUTTER_ROOT%\bin;%ANDROID_HOME%\platform-tools;%ANDROID_HOME%\cmdline-tools\latest\bin;%JAVA_HOME%\bin
